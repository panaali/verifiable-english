"""One-off: compare the et-3 Lean declarations with the approved blocks of spec 6.1 and 6.2 (whitespace-normalised,
theorems up to the proof's `:=`), using tools/check_readings_defaults_statements.declarations."""
import json, sys
from pathlib import Path
ROOT = Path('<worktree>')
sys.path.insert(0, str(ROOT / 'tools'))
from check_readings_defaults_statements import declarations, lean_of
spec = (ROOT / 'docs/spec/events-time-causation.md').read_text()
sec = spec.split('### 6.1 Definitions (ET-3)', 1)[1].split('### 6.3', 1)[0]
blocks = [b.split('```', 1)[0] for b in sec.split('```lean')[1:]]
approved = {}
for b in blocks:
    for k, v in declarations(b, statements_only=True).items():
        approved.setdefault(k, []).extend(v)
files = sorted((ROOT / 'lean/VE/Events').glob('*.lean'))
files = [f for f in files if f.name != 'Examples.lean']
lean = lean_of(files)
out = {'exact': [], 'differs': [], 'not_built': []}
for (kind, name), texts in sorted(approved.items()):
    found = lean.get((kind, name), [])
    if not found:
        out['not_built'].append(f'{kind} {name}')
        continue
    a, l = texts[0], found[0]
    if kind == 'def' and ':=' not in a:
        l = l.split(' :=', 1)[0]
    (out['exact'] if a == l else out['differs']).append(f'{kind} {name}' if a == l else
        {'decl': f'{kind} {name}', 'approved': a, 'lean': l})
out['counts'] = {k: len(v) for k, v in out.items() if isinstance(v, list)}
print(json.dumps(out, indent=1, ensure_ascii=False))
