import subprocess, sys, os, re, json
ROOT='<worktree>'
LEAN=ROOT+'/lean'
env=dict(os.environ, PATH=os.path.expanduser('~/.elan/bin')+':'+os.environ['PATH'])
MUTANTS={
 'drop-linear': ('VE/Events/Premises.lean',
   'def orderPremises (V : EventsEnv) : List VE.Formula := [irreflexiveF V, transitiveF V, linearF V]',
   'def orderPremises (V : EventsEnv) : List VE.Formula := [irreflexiveF V, transitiveF V]'),
 'reversed-date-facts': ('VE/Events/Premises.lean',
   '(consecutive V).map (fun p => ("events:date:" ++ isoPair p.1 p.2, dateFactF V p.1 p.2))',
   '((V.dates.flatMap fun a => V.dates.filterMap fun b => if b.num < a.num then some (a, b) else none).map\n    (fun p => ("events:date:" ++ isoPair p.1 p.2, dateFactF V p.1 p.2)))'),
 'no-gap-count': ('VE/Events/Calendar.lean',
   'decide (r p.2 - r p.1 - 1 > p.2.num - p.1.num - 1)', 'false'),
 'before-as-guard': ('VE/Events/Completion.lean',
   '| .binary p t w => !ng.contains p && (t == .var k || w == .var k)', '| .binary p t w => (t == .var k || w == .var k)'),
}
FILES=['VE/Events/Decl.lean','VE/Events/Premises.lean','VE/Events/Calendar.lean','VE/Events/Completion.lean','VE/Events/Examples.lean']
name=sys.argv[1]
path,old,new=MUTANTS[name]
src=open(os.path.join(LEAN,path)).read()
assert src.count(old)==1, name
open(os.path.join(LEAN,path),'w').write(src.replace(old,new))
try:
    b=subprocess.run(['lake','build','VE.Events.Examples'],cwd=LEAN,env=env,capture_output=True,text=True)
    errs=[l for l in b.stdout.splitlines() if 'error:' in l]
    # one scratch file: #guard lines still run when a proof upstream fails
    parts=['import VE.Semantics','import VE.Time.Calendar','import VE.Norms.Prefix','import VE.Finite']
    for f in FILES:
        parts+= [l for l in open(os.path.join(LEAN,f)).read().splitlines() if not l.startswith('import ')]
    scratch=os.path.join(LEAN,'EtThreeMutantScratch.lean')
    open(scratch,'w').write('\n'.join(parts)+'\n')
    s=subprocess.run(['lake','env','lean','EtThreeMutantScratch.lean'],cwd=LEAN,env=env,capture_output=True,text=True)
    os.remove(scratch)
    lines=open(os.path.join(LEAN,'VE/Events/Examples.lean')).read().splitlines()
    off=len(parts)-len([l for l in lines if not l.startswith('import ')])
    guards=[]
    for l in s.stdout.splitlines():
        m=re.match(r'EtThreeMutantScratch.lean:(\d+):\d+: error: (.*)',l)
        if m:
            n=int(m.group(1))
            if n>off:
                ex_line=n-off+sum(1 for x in lines if x.startswith('import '))
                guards.append((ex_line, lines[ex_line-1].strip()[:110], m.group(2)[:80]))
            else:
                guards.append((n,'(upstream)',m.group(2)[:80]))
    print(json.dumps({'mutant':name,'file':'lean/'+path,'lake_build_ok':b.returncode==0,
      'build_errors':[e[:200] for e in errs[:6]],'scratch_errors':guards[:12]},indent=1))
finally:
    subprocess.run(['git','checkout','--','lean/'+path],cwd=ROOT)
