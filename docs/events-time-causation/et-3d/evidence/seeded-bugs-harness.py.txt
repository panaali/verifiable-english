"""et-3d seeded bugs (et-3cb's harness with this item's mutants): apply one mutant to the committed tree, run
`lake build VE.Events.Examples`, then elaborate the six events files as one scratch file so every #guard and example
runs even when a proof upstream fails; restore. `test` names the declaration around each error.
Usage: python3 harness.py <worktree> (prints the JSON committed as ../mutants.json, unedited)."""
import subprocess, sys, os, re, json
ROOT = sys.argv[1]
LEAN = ROOT + '/lean'
env = dict(os.environ, PATH=os.path.expanduser('~/.elan/bin') + ':' + os.environ['PATH'])
D = 'VE/Events/Decl.lean'
K = 'VE/Events/Calendar.lean'
TYPED = 'D.properB && decide (D.year ≤ 9999) && E.constants (dateName D) == some V.day'
MUTANTS = [
 ('EventsTyped-bound-dropped', 'the year bound', D, TYPED, 'D.properB && E.constants (dateName D) == some V.day'),
 ('EventsTyped-bound-9998', 'the year bound', D, TYPED, TYPED.replace('9999', '9998')),
 ('EventsTyped-bound-10000', 'the year bound', D, TYPED, TYPED.replace('9999', '10000')),
 ('EventsTyped-bound-first-date-only', 'the year bound', D,
  'V.dates.all (fun D => ' + TYPED + ') &&',
  'V.dates.all (fun D => D.properB && E.constants (dateName D) == some V.day) &&\n'
  '  (V.dates.head?.all fun D => decide (D.year ≤ 9999)) &&'),
 ('WellFormedEvents-bound-dropped', 'the decoder side of the bound (regen_typed)', D,
  'D.properB && decide (D.year ≤ 9999) && E.constants (dateName D) == none',
  'D.properB && E.constants (dateName D) == none'),
 ('calendarCheck-after-bound-plus-one', 'embeddable_sound', K,
  '> dateMax - Dk.num then', '> dateMax + 1 - Dk.num then'),
 ('numbering-named-offset-plus-one', 'embeddable_sound', K,
  '| some D => D.num + (rank V M x - r D)', '| some D => D.num + (rank V M x - r D) + 1'),
]
FILES = [D, 'VE/Events/Premises.lean', K, 'VE/Events/Completion.lean', 'VE/Events/Conservative.lean',
         'VE/Events/Examples.lean']
IMPORTS = ['import VE.Semantics', 'import VE.Time.Calendar', 'import VE.Norms.Prefix', 'import VE.Finite',
           'import VE.Readings.Names', 'import VE.Readings.Superpose', 'import VE.Time.Wire']
HEAD = re.compile(r'(theorem|def|example|#guard|structure|abbrev)\b')

def run(name, claim, path, old, new):
    src = open(os.path.join(LEAN, path)).read()
    assert src.count(old) == 1, name
    open(os.path.join(LEAN, path), 'w').write(src.replace(old, new))
    try:
        b = subprocess.run(['lake', 'build', 'VE.Events.Examples'], cwd=LEAN, env=env, capture_output=True, text=True)
        parts, origin = list(IMPORTS), [None] * len(IMPORTS)
        for f in FILES:
            for i, l in enumerate(open(os.path.join(LEAN, f)).read().splitlines(), 1):
                if not l.startswith('import '):
                    parts.append(l); origin.append((f, i, l))
        scratch = os.path.join(LEAN, 'EtThreeDMutantScratch.lean')
        open(scratch, 'w').write('\n'.join(parts) + '\n')
        s = subprocess.run(['lake', 'env', 'lean', 'EtThreeDMutantScratch.lean'], cwd=LEAN, env=env,
                           capture_output=True, text=True)
        os.remove(scratch)
        failing, tests = [], []
        for l in s.stdout.splitlines():
            m = re.match(r'EtThreeDMutantScratch.lean:(\d+):\d+: error', l)
            if m:
                k = int(m.group(1)) - 1
                f, i, _ = origin[k]
                failing.append(f'lean/{f}:{i}')
                j = k
                while origin[j] is not None and not HEAD.match(origin[j][2]):
                    j -= 1
                t = origin[j][2].strip()[:110] if origin[j] is not None else '?'
                if t not in tests:
                    tests.append(t)
        return {'mutant': name, 'claim': claim, 'file': 'lean/' + path, 'language': 'Lean', 'replace': old, 'with': new,
                'lake_build_VE.Events.Examples': 'passes' if b.returncode == 0 else 'fails',
                'failing': failing[:12], 'test': tests[:8]}
    finally:
        subprocess.run(['git', 'checkout', '--', 'lean/' + path], cwd=ROOT)

head = subprocess.run(['git', 'rev-parse', '--short', 'HEAD'], cwd=ROOT, capture_output=True, text=True).stdout.strip()
out = {'harness': 'docs/events-time-causation/et-3d/evidence/seeded-bugs-harness.py.txt', 'tree': head,
       'mutants': [run(*m) for m in MUTANTS]}
subprocess.run(['lake', 'build', 'VE.Events.Examples'], cwd=LEAN, env=env, capture_output=True)
print(json.dumps(out, indent=1, ensure_ascii=False))
