import json, os, subprocess, sys
ROOT = "<worktree>"
env = dict(os.environ); env["PATH"] = os.path.expanduser("~/.elan/bin") + ":" + env["PATH"]
M = [
 ("lean/VE/Cardinality/Regions.lean", "inRegion reads `not b` as `b`",
  "  | .not b => !(b.inRegion ps r)", "  | .not b => b.inRegion ps r"),
 ("lean/VE/Cardinality/Erase.lean", "a proportion's erasure drops the nonempty-population conjunct",
  "  .and (.lt (.num dl 0) n) (p.cmp.erase (.scale dl p.den m) (.scale dl p.num n))",
  "  .and .top (p.cmp.erase (.scale dl p.den m) (.scale dl p.num n))"),
 ("lean/VE/Cardinality/Regions.lean", "the predicate limit fires at 6 instead of above 6",
  "    if (predsOn E fs s).length > maxPredicates then", "    if (predsOn E fs s).length ≥ maxPredicates then"),
 ("lean/VE/Cardinality/Syntax.lean", "a body atom needs only some declared predicate, of any arity or sort (specification and check together)",
  "  | .atom p => E.q.predicates p = some [.obj s]\n  | .any => True", "  | .atom p => (E.q.predicates p).isSome = true\n  | .any => True"),
 ("lean/VE/Cardinality/Semantics.lean", "`Cmp.holds .ge` compares the wrong way",
  "  | .ge, a, b => b ≤ a", "  | .ge, a, b => a ≤ b"),
 ("lean/VE/Cardinality/Erase.lean", "countFin counts listed objects with duplicates",
  "  ((firstUse (M.domains s)).filter (fun x => b.evalFin M x)).length", "  ((M.domains s).filter (fun x => b.evalFin M x)).length"),
]
extra = {3: [("lean/VE/Cardinality/Syntax.lean", "  | .atom p => E.q.predicates p == some [.obj s]", "  | .atom p => (E.q.predicates p).isSome")]}
out = []
for i, (f, what, old, new) in enumerate(M):
    edits = [(f, old, new)] + extra.get(i, [])
    saved = {}
    for (g, o, n) in edits:
        p = os.path.join(ROOT, g); s = saved.get(p) or open(p).read(); saved.setdefault(p, s)
        cur = open(p).read(); assert o in cur, (g, o); open(p, "w").write(cur.replace(o, n, 1))
    r = subprocess.run(["lake", "build", "VE.Cardinality.Examples"], cwd=ROOT + "/lean", env=env, capture_output=True, text=True)
    for p, s in saved.items(): open(p, "w").write(s)
    errs = [l for l in r.stdout.splitlines() if "error" in l]
    out.append({"file": f, "mutation": what, "build_exit": r.returncode, "first_errors": errs})
    print(json.dumps(out[-1], ensure_ascii=False), flush=True)
json.dump(out, open("/tmp/qta1/mutants-run.json", "w"), indent=1, ensure_ascii=False)
