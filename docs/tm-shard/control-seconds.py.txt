"""Time the control section of trust_mutants.main (scratch copy, control build, probe install, check probes) for the
mutants part K/N owns (0/0: the whole run), as main does it."""
import json, os, sys, tempfile, time
from concurrent.futures import ThreadPoolExecutor
from pathlib import Path
sys.path.insert(0, "/home/seyran/code/ve-eff-tmshard/tools")
import trust_mutants as tm
root, base, part = Path(sys.argv[1]), sys.argv[2], sys.argv[3]
k, n = map(int, part.split("/"))
found, tree, fatal = tm.plan(root, base, tm.TRUSTED)
todo = [m for m in found[:30] if tm.owns(m.id, (k, n) if n else None)]
env = {**os.environ, "PATH": f"{Path.home() / '.elan/bin'}{os.pathsep}{os.environ.get('PATH', '')}"}
with tempfile.TemporaryDirectory(prefix="trust-mutants-") as d:
    t0 = time.monotonic()
    first = tm.Scratch(root / "lean", Path(d))
    t1 = time.monotonic()
    built = sorted({x for m in todo for x in tm.targets(tree, m, [], False)})
    code, _ = first.lake(env, "build", *(f"+{x}" for x in built))
    t2 = time.monotonic()
    files = {f: (first.lean / f, tree.roles[f]) for f in sorted({tree.files[x] for x in built})
             if any(r == "check" for _, _, r, _ in tree.roles.get(f, ()))}
    assert tm.install(lambda *a: first.lake(env, *a), first.lean)
    with ThreadPoolExecutor(max_workers=4) as ex:
        list(ex.map(lambda kv: tm.probe(first.lake, env, kv[1][0].read_text(), kv[1][1], tm.beside(first.lean, kv[0]),
                                        False), files.items()))
    t3 = time.monotonic()
print(json.dumps({"part": part, "mutants": len(todo), "targets": len(built), "check_files": len(files), "code": code,
                  "copy_s": round(t1 - t0), "build_s": round(t2 - t1), "probe_s": round(t3 - t2),
                  "control_s": round(t3 - t0), "load": os.getloadavg()[0]}))
