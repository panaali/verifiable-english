"""kb-4a: KB-4's module measure (docs/kernel/kb-4/MEASURE.md, kb-8b) regenerated after the acts hook.

The same method as kb-8b's script (embedded in docs/kernel/kb-4/MEASURE.md): `tools/kernel_closure.py`'s `why()` on every
module a kernel path counts, its per-part figures asserted equal to the baseline's, then the 25 largest modules' five
largest definitions with one shortest chain each. It also renders the Markdown tables (kb-8b printed them by hand from
its JSON). Usage: python3 measure.py ROOT OUT.md
"""
import json
import sys
from pathlib import Path

ROOT = Path(sys.argv[1])
sys.path.insert(0, str(ROOT / "tools"))
import kernel_closure as kc  # noqa: E402

registry = json.loads((ROOT / "tools/kernel-paths.json").read_text())
baseline = json.loads((ROOT / "docs/kernel/closure-baseline.json").read_text())
kernel = [p["name"] for p in registry["paths"] if p.get("tier") == "kernel"]
modules = sorted({m for row in baseline["paths"] if row["name"] in kernel
                  for m in list(row["spec_modules"]) + list(row["wrap_modules"])})
found = kc.why(registry, modules)
for row in baseline["paths"]:
    if row["name"] not in kernel:
        continue
    for part in ("spec", "wrap"):
        for m, n in row[part + "_modules"].items():
            mine = [r for r in found[m] if r["path"] == row["name"] and r["part"] == part]
            assert len(mine) == 1 and mine[0]["lines"] == n, (row["name"], part, m, n, mine)
out = {}
for m, rows in found.items():
    rng = lambda parts: [(m, d["from"], d["to"]) for r in rows if r["part"] in parts for d in r["definitions"]]  # noqa: E731
    defs = {}
    for r in rows:
        for d in r["definitions"]:
            defs[d["name"]] = d["lines"]
    parts = {}
    for r in rows:
        parts.setdefault(r["path"], []).append(r["part"])
    out[m] = {"spec": kc.covered(rng({"spec"})), "wrap": kc.covered(rng({"wrap"})),
              "total": kc.covered(rng({"spec", "wrap"})), "parts": parts,
              "defs": sorted(defs.items(), key=lambda kv: (-kv[1], kv[0]))}
total = sum(v["total"] for v in out.values())
assert total == baseline["kernel_lines"], (total, baseline["kernel_lines"])
top = sorted(out, key=lambda m: (-out[m]["total"], m))[:25]
names = [d for m in top for d, _ in out[m]["defs"][:5]]
chains = kc.why(registry, names)
best = {}
for d in names:
    rows = [r for r in chains[d] if r["chain"]]
    rows.sort(key=lambda r: (len(r["chain"]), kernel.index(r["path"]), r["part"] != "spec"))
    best[d] = rows[0]

ACTS = lambda m: m.startswith("VE.Acts.") or m == "VE.Grammar.Table"  # noqa: E731
GATE = {"VE.Compose.Dispatch", "VE.Service.Standalone", "VE.Order.ProfileGate"}
WIRE = {"VE.Route.Wire", "VE.Route.Decode", "VE.Route.PyDecoder", "VE.Norms.Wire", "VE.Bridge", "VE.Core.Bridge",
        "VE.CertificateWire", "VE.Order.Wire"}
LOGIC = ("VE.Order", "VE.Core", "VE.Closure", "VE.Route", "VE.Norms", "VE.Definitions", "VE.Semantics", "VE.Syntax",
         "VE.Tableau", "VE.TableauChecker", "VE.TableauClosure", "VE.Finite", "VE.Paths")


def group(m):
    if ACTS(m):
        return "acts decoding"
    if m in GATE:
        return "profile gate and dispatch"
    if m in WIRE:
        return "wire codecs"
    if m.startswith("VE.SoftNorms."):
        return "soft norms"
    if any(m == p or m.startswith(p + ".") for p in LOGIC):
        return "logic and semantics"
    return "other"


GROUPS = ["acts decoding", "profile gate and dispatch", "wire codecs", "logic and semantics", "soft norms", "other"]
f = "{:,}".format
paths_of = lambda m: ", ".join(f"{p} ({', '.join(ps)})" for p, ps in sorted(  # noqa: E731
    out[m]["parts"].items(), key=lambda kv: kernel.index(kv[0])))
lines = []
w = lines.append
w("# KB-4's measure after kb-4a: the 25 largest modules of the kernel tier")
w("")
w("Regenerated by kb-4a (branch `feature/kb-4a`) after the acts hook, by the method of [the kb-8b measure](../kb-4/MEASURE.md)")
w("(which stays as the measure before kb-4a, on main `9a3fd7a9f9`): `tools/kernel_closure.py`'s `why()` on every module a")
w(f"kernel path counts, its per-path, per-part figures asserted equal to [closure-baseline.json](../closure-baseline.json)'s; "
  f"the {len(out)} modules' lines add up to {f(total)}, the baseline's `kernel_lines`. Kernel tier: the paths "
  + ", ".join(f"`{k}`" for k in kernel) + ". The groups, the columns and the chains are as in the kb-8b measure. The script is at the end.")
w("")
w("## Groups")
w("")
w("| Group | Modules listed below | Their lines | All kernel modules of the group | Their lines |")
w("|---|---:|---:|---:|---:|")
for g in GROUPS:
    ms = [m for m in out if group(m) == g]
    ts = [m for m in top if group(m) == g]
    w(f"| {g} | {len(ts)} | {f(sum(out[m]['total'] for m in ts))} | {len(ms)} | {f(sum(out[m]['total'] for m in ms))} |")
w(f"| **total** | **{len(top)}** | **{f(sum(out[m]['total'] for m in top))}** | **{len(out)}** | **{f(total)}** |")
w("")
w("## Acts decoding: what remains in the kernel")
w("")
w("Every module of the group that a kernel path still counts, with one shortest chain per definition (kb-4a's REPORT")
w("reads this table).")
w("")
acts = sorted((m for m in out if group(m) == "acts decoding"), key=lambda m: (-out[m]["total"], m))
w("| Module | Lines | Spec lines | Wrap lines | Paths that count it |")
w("|---|---:|---:|---:|---|")
for m in acts:
    w(f"| `{m}` | {out[m]['total']} | {out[m]['spec']} | {out[m]['wrap']} | {paths_of(m)} |")
w("")
adefs = [d for m in acts for d, _ in out[m]["defs"]]
achains = kc.why(registry, adefs)
w("| Module | Definition | Lines | Chain (path, part: root first) |")
w("|---|---|---:|---|")
for m in acts:
    for d, n in out[m]["defs"]:
        rows = sorted((r for r in achains[d] if r["chain"]),
                      key=lambda r: (len(r["chain"]), kernel.index(r["path"]), r["part"] != "spec"))
        r = rows[0]
        w(f"| `{m}` | `{d}` | {n} | {r['path']} {r['part']}: " + " → ".join(f"`{c}`" for c in r["chain"]) + " |")
for g in GROUPS:
    ts = [m for m in top if group(m) == g]
    ms = [m for m in out if group(m) == g]
    w("")
    w(f"## {g[0].upper() + g[1:]}")
    w("")
    w(f"Listed: {len(ts)} of the 25, {f(sum(out[m]['total'] for m in ts))} lines; the group in the whole kernel tier: "
      f"{len(ms)} modules, {f(sum(out[m]['total'] for m in ms))} lines.")
    w("")
    if not ts:
        w("None of the 25 largest modules is in this group.")
        continue
    w("| Module | Lines | Spec lines | Wrap lines | Paths that count it |")
    w("|---|---:|---:|---:|---|")
    for m in ts:
        w(f"| `{m}` | {out[m]['total']} | {out[m]['spec']} | {out[m]['wrap']} | {paths_of(m)} |")
    w("")
    w("| Module | Definition | Lines | Chain (path, part: root first) |")
    w("|---|---|---:|---|")
    for m in ts:
        for d, n in out[m]["defs"][:5]:
            r = best[d]
            w(f"| `{m}` | `{d}` | {n} | {r['path']} {r['part']}: " + " → ".join(f"`{c}`" for c in r["chain"]) + " |")
w("")
w("## The script")
w("")
w("`python3 measure.py <worktree> docs/kernel/kb-4a/MEASURE.md` (needs a built `lean/.lake`; about a minute):")
w("")
w("```python")
w(Path(__file__).read_text().rstrip("\n"))

print("modules", len(out), "total", total, "acts", sum(out[m]["total"] for m in acts))
