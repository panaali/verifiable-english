out = []
ONLY = sys.argv[1:]
for bug, rel, old, new, tests in PLANTS:
    if ONLY and not bug.startswith(ONLY[0]):
        continue
    path = ROOT / rel
    text = path.read_text()
    assert text.count(old) == 1, (bug, text.count(old))
    path.write_text(text.replace(old, new))
    try:
        run = subprocess.run(["uv", "run", "pytest", "-q", "-p", "no:cacheprovider", *tests], cwd=ROOT, capture_output=True, text=True, timeout=1800)
    finally:
        path.write_text(text)
    lines = [l for l in run.stdout.splitlines() if l.startswith("FAILED") or " failed" in l or " passed" in l]
    killed = run.returncode != 0 and any(l.startswith("FAILED") for l in lines)
    out.append({"bug": bug, "file": rel, "test": tests[0], "killed": killed, "output": [l[:200] for l in lines][-4:]})
    print(("KILLED  " if killed else "SURVIVED ") + bug, flush=True)
saved = json.loads((ROOT / "docs/infra/fid-mech-b4/seeded-bugs.json").read_text()) if ONLY else []
for o in out:
    saved = [x for x in saved if not x["bug"].startswith(ONLY[0])] if ONLY else saved
merged = saved + out if ONLY else out
order = [p[0] for p in PLANTS]
merged.sort(key=lambda x: order.index(x["bug"]))
(ROOT / "docs/infra/fid-mech-b4/seeded-bugs.json").write_text(json.dumps(merged, indent=1) + "\n")
