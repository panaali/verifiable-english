out = []
for sid, f, what, old, new, test in SEEDS:
    p = ROOT / f
    src = p.read_text()
    assert src.count(old) == 1, (sid, src.count(old))
    p.write_text(src.replace(old, new))
    try:
        r = subprocess.run(["uv", "run", "pytest", "-q", "-x", test], cwd=ROOT, capture_output=True, text=True)
    finally:
        p.write_text(src)
    failed = [l for l in r.stdout.splitlines() if l.startswith("FAILED")]
    out.append({"id": sid, "file": f, "bug": what, "test": test, "killed": r.returncode != 0, "failed": failed[:3]})
    print(sid, r.returncode != 0, failed[:1] or r.stdout.strip().splitlines()[-1:])
Path(sys.argv[1]).write_text(json.dumps(out, indent=1) + "\n")
