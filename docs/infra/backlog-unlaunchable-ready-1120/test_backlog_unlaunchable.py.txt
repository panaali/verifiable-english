def write(rel, text):
    os.makedirs(os.path.dirname(f"{D}/{rel}"), exist_ok=True)
    with open(f"{D}/{rel}", "w") as f:
        f.write(text)


def row(i, kind="infra", owner="", status="proposed", task=None, ready_h=None, **links):
    """A toy row; task = the task file's text (prompts/task-<i>.txt), None = no links.task."""
    r = bl.blank(i, kind, f"title {i}", priority=50, owner=owner, status=status)
    if task is not None:
        write(f"prompts/task-{i}.txt", task)
        r["links"]["task"] = f"prompts/task-{i}.txt"
    r["links"].update(links)
    if ready_h is not None:
        r["ready_t"] = bl.utc(bl.NOW_T - ready_h * 3600)
    return r


def backlog(*rows):
    write("BACKLOG.jsonl", "".join(json.dumps(r, sort_keys=True) + "\n" for r in rows))


def raw():
    return open(f"{D}/BACKLOG.jsonl").read()


def cli(*args):
    p = subprocess.run([sys.executable, "-I", BL, *args], capture_output=True, text=True, env=ENV, timeout=600)
    return p.returncode, p.stdout, p.stderr

class Refused(unittest.TestCase):
    def setUp(self):
        shutil.rmtree(D, True)
        os.makedirs(D)

    def check_refused(self, rc, out, err, want):
        self.assertEqual(rc, 1, (out, err))
        lines = err.strip().splitlines()
        self.assertEqual(len(lines), 1, err)                       # one line
        self.assertTrue(lines[0].startswith("refused: "), err)
        self.assertIn(want, lines[0])

    def test_add_status_ready_refused_for_each_pair(self):
        for n, (kind, owner, task, links, want) in enumerate(REFUSED):
            with self.subTest(kind=kind, owner=owner, task=task):
                backlog(row("other"))
                before = raw()
                if task is not None:
                    write(f"prompts/task-r{n}.txt", task)
                args = ["add", f"r{n}", kind, "a", "title", "--status", "ready", "--owner", owner]
                args += ["--link", f"task=prompts/task-r{n}.txt"] if task is not None else []
                args += [x for k, v in links.items() for x in ("--link", f"{k}={v}")]
                self.check_refused(*cli(*args), want)
                self.assertEqual(raw(), before)                    # row not added, file unchanged

    def test_update_status_and_update_refused_row_unchanged(self):
        for n, (kind, owner, task, links, want) in enumerate(REFUSED):
            with self.subTest(kind=kind, owner=owner, task=task):
                backlog(row(f"r{n}", kind, owner, "blocked", task, **links))
                before = raw()
                self.check_refused(*cli("update-status", f"r{n}", "ready"), want)
                self.assertEqual(raw(), before)
                self.check_refused(*cli("update", f"r{n}", "--status", "ready", "--note", "go"), want)
                self.assertEqual(raw(), before)

    def test_update_that_breaks_a_ready_rows_launcher_is_refused(self):
        backlog(row("a", "infra", "infra feed", "ready", "x"))
        before = raw()
        self.check_refused(*cli("update", "a", "--owner", "infra planner"), "'infra planner' has no launcher")
        self.assertEqual(raw(), before)
        rc, out, err = cli("update", "a", "--owner", "DRI decider")          # another launcher: fine
        self.assertEqual(rc, 0, err)

    def test_rows_already_ready_are_not_rewritten_and_lint_lists_them(self):
        backlog(row("old", "infra", "feature planner", "ready", "x"), row("fine", "infra", "", "ready", "x"),
                row("held", "feature", "", "blocked", NOPATHS), row("dec", "decision", "DRI", "ready"))
        rc, out, err = cli("update", "old", "--note", "still waiting")       # a note on an already-refused ready row passes
        self.assertEqual(rc, 0, err)
        self.assertEqual(bl.load()["old"]["status"], "ready")
        rc, out, err = cli("lint")
        self.assertEqual(rc, 1, err)
        ls = [l for l in out.splitlines() if not l.startswith("#")]
        self.assertEqual(len(ls), 1, out)
        self.assertTrue(ls[0].startswith("old\tready\tinfra\tfeature planner\towner 'feature planner' has no launcher"), ls)
        backlog(row("fine", "infra", "", "ready", "x"))
        rc, out, err = cli("lint")
        self.assertEqual(rc, 0, out + err)


class Allowed(unittest.TestCase):
    def setUp(self):
        shutil.rmtree(D, True)
        os.makedirs(D)

    def test_each_allowed_pair_is_accepted_by_every_command(self):
        for n, (kind, owner, task, links, who) in enumerate(ALLOWED):
            with self.subTest(kind=kind, owner=owner):
                r = row(f"a{n}", kind, owner, "ready", task, **links)
                self.assertEqual(bl.start_rule(r)[0], who)
                self.assertEqual(bl.ready_refusal(r), "")
                backlog()
                args = ["add", f"a{n}", kind, "t", "--status", "ready", "--owner", owner]
                args += ["--link", f"task=prompts/task-a{n}.txt"] if task is not None else []
                args += [x for k, v in links.items() for x in ("--link", f"{k}={v}")]
                rc, out, err = cli(*args)
                self.assertEqual((rc, out.strip()), (0, f"add a{n}: ready"), err)
                backlog(dict(r, status="proposed"))
                self.assertEqual(cli("update-status", f"a{n}", "ready")[0], 0)
                self.assertEqual(cli("update", f"a{n}", "--status", "ready")[0], 0)
                self.assertEqual(bl.load()[f"a{n}"]["status"], "ready")

    def test_answered_rows_are_not_launched_and_not_refused(self):
        for kind, owner in NOT_LAUNCHED:
            with self.subTest(kind=kind):
                backlog(row("q", kind, owner))
                self.assertEqual(cli("update-status", "q", "ready")[0], 0)

    def test_other_statuses_are_never_refused(self):
        backlog(row("p", "infra", "infra planner", "ready", "x"))
        for st in ("blocked", "in-flight", "done"):
            self.assertEqual(cli("update-status", "p", st)[0], 0, st)


class Seed(unittest.TestCase):
    def setUp(self):
        shutil.rmtree(D, True)
        os.makedirs(D)

    def test_blocked_row_turning_ready_by_itself_is_refused(self):
        dep = row("dep", status="merged")
        bad = row("bad", "infra", "feature planner", "blocked", "x")
        nopaths = row("np", "feature", "", "blocked", NOPATHS)
        good = row("good", "infra", "", "blocked", "x")
        for r in (bad, nopaths, good):
            r["blocked_by"] = ["dep"]
        backlog(dep, bad, nopaths, good)
        bl.seed(quiet=True)
        rows = bl.load()
        self.assertEqual(rows["good"]["status"], "ready")
        for i, want in (("bad", "'feature planner' has no launcher"), ("np", "has no 'Paths:' line")):
            self.assertEqual(rows[i]["status"], "blocked")
            self.assertTrue(rows[i]["state"].startswith("blockers merged; not set ready: "), rows[i]["state"])
            self.assertIn(want, rows[i]["state"])
            self.assertNotIn("ready_t", rows[i])
        before = {i: rows[i] for i in ("bad", "np")}
        bl.seed(quiet=True)                                       # idempotent: the held rows are not rewritten
        self.assertEqual({i: bl.load()[i] for i in ("bad", "np")}, before)
        rc, out, _ = cli("lint")
        self.assertEqual(rc, 1)
        self.assertIn("bad\tblocked\tinfra\tfeature planner\towner 'feature planner' has no launcher", out)


class InfraFeedReadsTheTable(unittest.TestCase):
    def test_item_reason_names_the_missing_launcher(self):
        rows = {r["id"]: r for r in (row("fp", "infra", "feature planner", "ready", "x"), row("ok", "infra", "", "ready", "x"),
                                      row("dg", "infra", "delegate dri-x", "ready", "x"))}

        self.assertIn("'feature planner' has no launcher for kind infra", why["fp"])
        self.assertEqual(why["ok"], "")
        self.assertEqual(why["dg"], "owner 'delegate dri-x' starts it, not this feed")

class Unlaunched(unittest.TestCase):
    def setUp(self):
        shutil.rmtree(D, True)
        os.makedirs(D)

    def rows(self):
        rs = [row("old-bad", "infra", "infra planner", "ready", "x", ready_h=2),
              row("old-ok", "infra", "", "ready", "x", ready_h=3),
              row("young", "infra", "infra planner", "ready", "x", ready_h=0.5),
              row("running", "infra", "", "ready", "x", ready_h=5, job="running"),
              row("deleg", "infra", "delegate dri-d", "ready", "x", ready_h=5, delegate="dri-d"),
              row("queued", "feature", "", "ready", PATHS, ready_h=5),
              row("blocked", "infra", "infra planner", "blocked", "x", ready_h=5)]
        rs[1]["state"] = "feed: account room"
        write("FEATURE-QUEUE.tsv", f"5\tqueued\tfeature/queued\t{D}/prompts/task-queued.txt\t-\t32G\tlean/VE/X.lean\n")
        return {r["id"]: r for r in rs}

    def test_lines_id_age_owner_why(self):
        rows = self.rows()
        got = bl.unlaunched_lines(rows, {}, {}, {"dri-d"}, {"running"})
        self.assertEqual(len(got), 2, got)
        self.assertEqual(got[0], "old-bad, 2.0h, infra planner: " + bl.short(bl.ready_refusal(rows["old-bad"]), 200))
        self.assertIn("'infra planner' has no launcher for kind infra", got[0])
        self.assertEqual(got[1], "old-ok, 3.0h, -: infra feed: not started (state: feed: account room)")
        self.assertEqual(len(bl.unlaunched_lines(rows, {}, {}, set(), set())), 4)     # the jobs gone: running and deleg listed

    def test_digest_lists_them(self):
        backlog(*self.rows().values())
        write("inbox/orchestrator-digest.md", "1794000000\t2026-11-20T10:00Z\tinbox/orchestrator/x.md\tunmarked\tnote\n")
        p = subprocess.run([sys.executable, "-I", CD, "digest-take"], capture_output=True, text=True, env=ENV, timeout=900)
        self.assertEqual(p.returncode, 0, p.stderr)
        out = p.stdout.splitlines()
        i = next(n for n, l in enumerate(out) if l.startswith("== ready > 60 min with no live launch: "))
        # running/deleg are not live in this test (no such job), queued is in FEATURE-QUEUE.tsv
        self.assertEqual(out[i], "== ready > 60 min with no live launch: 4 (backlog.sh unlaunched; refused pairs: backlog.sh lint)")
        ids = [l[2:].split(",")[0] for l in out[i + 1:i + 5]]
        self.assertEqual(ids, ["deleg", "old-bad", "old-ok", "running"])
        self.assertTrue(out[i + 2].startswith("- old-bad, 2.0h, infra planner: owner 'infra planner' has no launcher"), out)
        self.assertLess(next(n for n, l in enumerate(out) if l.startswith("== digest (")), i)   # after the notices
